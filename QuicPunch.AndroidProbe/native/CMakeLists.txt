cmake_minimum_required(VERSION 3.20)
project(qp_transport C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(MSQUIC_ROOT "" CACHE PATH "Pinned MsQuic source tree")
set(MSQUIC_LIBRARY "" CACHE FILEPATH "Pinned MsQuic library")
set(OPENSSL_ROOT "" CACHE PATH "Pinned OpenSSL installation")
add_library(qp_transport SHARED qp_transport.c)
add_executable(qp_network_probe qp_network_probe.c)
target_link_libraries(qp_network_probe PRIVATE qp_transport)
if(MSVC)
  target_compile_definitions(qp_network_probe PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
target_include_directories(qp_transport PRIVATE "${MSQUIC_ROOT}/src/inc" "${OPENSSL_ROOT}/include")
if(WIN32)
  target_compile_definitions(qp_transport PRIVATE QUIC_PLATFORM_WINDOWS _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(qp_transport PRIVATE "${MSQUIC_LIBRARY}" "${OPENSSL_ROOT}/lib/libcrypto.lib" ws2_32 crypt32 bcrypt advapi32 user32)
else()
  target_link_libraries(qp_transport PRIVATE "${MSQUIC_LIBRARY}" "${OPENSSL_ROOT}/lib/libcrypto.a" pthread dl)
  set_target_properties(qp_transport PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
